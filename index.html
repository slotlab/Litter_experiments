async function captureAndOCR(mode) { if (!worker) return; if (!video.videoWidth || !video.videoHeight) { statusEl.textContent = "Camera not ready — wait and try again"; return; } const isMass = mode === "mass"; canvas.width = video.videoWidth; canvas.height = video.videoHeight; ctx.drawImage(video, 0, 0); const cw = Math.floor(canvas.width * 0.7); const ch = Math.floor(canvas.height * 0.4); const crop = document.createElement("canvas"); crop.width = cw; crop.height = ch; crop.getContext("2d").drawImage( canvas, (canvas.width - cw) / 2, (canvas.height - ch) / 2, cw, ch, 0, 0, cw, ch ); const processed = preprocess(crop, !isMass); preview.src = processed.toDataURL("image/jpeg", 0.85); statusEl.textContent = "Reading…"; rawEl.textContent = ""; try { let bestText = ""; let bestScore = -1; // Try a couple of page-segmentation modes and keep the best result const psms = isMass ? ["7", "8"] : ["7", "6", "8"]; for (const psm of psms) { await worker.setParameters({ tessedit_pageseg_mode: psm, tessedit_char_whitelist: isMass ? "0123456789." : "" }); const { data: { text } } = await worker.recognize(processed); const cleaned = (text || "").replace(/\s+/g, " ").trim(); const score = cleaned.length + (cleaned.match(/[QAqa]\d{5,6}/i) ? 10 : 0); if (score > bestScore) { bestScore = score; bestText = cleaned; } } rawEl.textContent = "Raw OCR: " + (bestText || "(nothing)"); if (mode === "tag") { // Prefer a clean match let match = bestText.match(/[QAqa]\d{5,6}/i); if (!match) { // Soft cleanup of common OCR mistakes then retry const fixed = bestText .replace(/[Oo]/g, "0") .replace(/[IiLl]/g, "1") .replace(/[Ss]/g, "5") .replace(/[Zz]/g, "2"); match = fixed.match(/[QAqa]\d{5,6}/i); if (match) bestText = fixed; } if (match) { bagInput.value = match[0].toUpperCase(); } else { const tokens = bestText.replace(/[^A-Za-z0-9]/g, " ").split(/\s+/).filter(Boolean); const best = tokens.sort((a, b) => b.length - a.length)[0] || ""; bagInput.value = best.toUpperCase(); } } else { const match = bestText.match(/\d+\.?\d*/); massInput.value = match ? match[0] : bestText.replace(/[^0-9.]/g, ""); } statusEl.textContent = "OCR done — edit if needed"; } catch (err) { statusEl.textContent = "OCR error: " + err.message; statusEl.className = "status err"; } }
